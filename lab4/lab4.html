# Завдання 4. Розкладання на складові
from statsmodels.tsa.seasonal import STL
stl = STL(bread['ціна'], period=12)
res = stl.fit()
res.plot()
T = res.trend
S = res.seasonal
R = res.resid


var_R = np.var(R)
var_T_R = np.var(T + R)
var_S_R = np.var(S + R)

strength_trend = max(0.0, 1.0 - (var_R / var_T_R))
strength_seasonal = max(0.0, 1.0 - (var_R / var_S_R))

print(f"Сила тренду (Ft):     {strength_trend:.4f}")
print(f"Сила сезонності (Fs): {strength_seasonal:.4f}")

def calculate_metrics(y_true, y_pred):
    mae = np.mean(np.abs(y_true - y_pred))
    rmse = np.sqrt(np.mean((y_true - y_pred)**2))
    mape = np.mean(np.abs((y_true - y_pred) / y_true)) * 100
    return mae, rmse, mape

    before = bread['ціна'][:k]

    t_k = (before.mean() - after.mean())/math.sqrt(np.var(before)/len(before) + np.var(after)/len(after))
    t[k] = math.fabs(t_k)

Виконується самостійно.
